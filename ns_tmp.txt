const SS = SpreadsheetApp.getActiveSpreadsheet();
var QZ_TZ = 'Asia/Seoul';
function qzTodayKST_(){ return Utilities.formatDate(new Date(), QZ_TZ, 'yyyy-MM-dd'); }
function stDateStr_(v){ if (v instanceof Date) return Utilities.formatDate(v, QZ_TZ, 'yyyy-MM-dd'); return String(v || '').trim(); }

function doGet(e){
  var action = (e && e.parameter && e.parameter.action) || '';
  if(action === 'psSubmit') return psSubmit_(e);
  if(action === 'psState')  return psState_(e);
  if(action === 'psSetup')  return psSetup_(e);
  if(action === 'psPick')   return psPick_(e);
  if(action === 'psAdmin')  return psAdmin_(e);
  if(action === 'psVote')   return psVote_(e);
  if(action === 'psGuess')  return psGuess_(e);
  if(action === 'psMyPeer') return psMyPeer_(e);
  if(action === 'psRoster') return psRosterPub_(e);
  if(action === 'pbBoard')  return pbBoard_(e);
  if(action === 'pbTeams')  return pbTeams_(e);
  if(action === 'pbRound')  return pbRound_(e);
  return ContentService.createTextOutput(JSON.stringify({error:'unknown action'})).setMimeType(ContentService.MimeType.JSON);
}

/* ═══════════════ 프리시즌: 축구 성향 검사 + 3주장 블라인드 드래프트 ═══════════════
 * 시트: 프리시즌_성향 [제출일시, 이름, 유형코드, 점수JSON, 키퍼가능, 응답]
 *  - 참가명단(PS_ROSTER)을 운영자가 등록하면 명단에 있는 사람만 검사 제출·드래프트 대상
 *  - 1인 1회 제출 (재검사가 필요하면 운영자가 psAdmin op=delone 으로 그 사람 결과 삭제)
 * 드래프트 상태: ScriptProperties 'PS_DRAFT' (JSON)
 *  - 주장 3명(신청자 풀에서 제외) · 스네이크 순서 1-2-3-3-2-1 · 주장별 PIN
 *  - 선수 이름은 공개(reveal) 전까지 운영자 외에는 내려가지 않음 → 완전 블라인드
 */
var PS_SHEET = '프리시즌_성향';
var PS_ADMIN_PW = '__PW__';            // 운영자 비밀번호 (변경 권장)
var PS_KEY = 'PS_DRAFT';

function psOut_(o){ return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function psSheet_(){
  var sh = SS.getSheetByName(PS_SHEET);
  if(!sh){ sh = SS.insertSheet(PS_SHEET); sh.appendRow(['제출일시','이름','유형코드','점수','키퍼가능','응답']); }
  return sh;
}
function psEntries_(){
  var v = psSheet_().getDataRange().getValues().slice(1), map = {};
  v.forEach(function(r){
    var name = String(r[1]||'').trim(); if(!name) return;
    var sc = {}; try{ sc = JSON.parse(r[3]||'{}'); }catch(e){}
    map[name] = { name:name, code:String(r[2]||''), scores:sc, gk: r[4]===true || String(r[4]).toUpperCase()==='TRUE' || r[4]==='O', ans:String(r[5]||'') };
  });
  return Object.keys(map).map(function(k){ return map[k]; });
}
function psLoad_(){ try{ return JSON.parse(PropertiesService.getScriptProperties().getProperty(PS_KEY)||'null'); }catch(e){ return null; } }
function psSave_(d){ PropertiesService.getScriptProperties().setProperty(PS_KEY, JSON.stringify(d)); }
function psRoster_(){ try{ return JSON.parse(PropertiesService.getScriptProperties().getProperty('PS_ROSTER')||'[]'); }catch(e){ return []; } }
function psSnake_(n, teams){ var o=[]; for(var r=0; o.length<n; r++){ for(var i=0;i<teams && o.length<n;i++){ o.push(r%2===0 ? i : teams-1-i); } } return o; }

/* ═══ 동료 추천: "우리 팀의 OO는 누구?" — 6개 포지션형 유형마다 2명까지 추천 (본인 제외 · 익명 · 득표수 비공개) ═══
 * 시트: 프리시즌_동료추천 [일시, 날짜, 투표자, 유형, 추천1, 추천2]  — 같은 날 다시 내면 그날 것만 덮어씀, 주가 바뀌면 누적
 * 한 유형에서 PV_MIN 표 이상 받으면 카드에 '동료 추천' 배지 (여러 유형이면 가장 많이 받은 것 하나)
 */
var PV_SHEET = '프리시즌_동료추천', PV_CATS = ['ST','PM','BW','SW','BB','CP'], PV_MIN = 3;
function pvSheet_(){
  var sh = SS.getSheetByName(PV_SHEET);
  if(!sh){ sh = SS.insertSheet(PV_SHEET); sh.appendRow(['일시','날짜','투표자','유형','추천1','추천2']); sh.getRange('B:B').setNumberFormat('@'); sh.setFrozenRows(1); }
  return sh;
}
function pvTally_(){
  var sh = SS.getSheetByName(PV_SHEET), t = {}; if(!sh) return t;
  sh.getDataRange().getValues().slice(1).forEach(function(r){
    var cat = String(r[3]||'').trim(); if(PV_CATS.indexOf(cat) < 0) return;
    [r[4], r[5]].forEach(function(n){ n = String(n||'').trim(); if(!n) return; (t[n] = t[n] || {})[cat] = ((t[n]||{})[cat] || 0) + 1; });
  });
  return t;
}
// 배지 문자열: 포지션(가장 많이 받은 것 · 동률이면 '+'로 모두) 뒤에, 캡틴은 따로 '|CP' — 예) 'ST' · 'ST+SW' · 'SW|CP' · '|CP'
function pvBadge_(t, name){
  var m = t[name]; if(!m) return '';
  var top = 0, pos = [];
  PV_CATS.forEach(function(c){ if(c !== 'CP' && (m[c]||0) > top) top = m[c]; });
  if(top >= PV_MIN) PV_CATS.forEach(function(c){ if(c !== 'CP' && m[c] === top) pos.push(c); });
  var cp = (m.CP||0) >= PV_MIN;
  return pos.join('+') + (cp ? '|CP' : '');
}
/* ═══ MY GUESS 점수판: 이름 맞추기 결과를 서버에 쌓음 ═══
 * 시트: 프리시즌_추측 [일시, 드래프트 날짜, 이름, 맞춘 수, 전체] — 드래프트마다 한 사람 한 번(처음 낸 것만)
 * 점수는 서버가 직접 채점 (보낸 추측과 실제 카드 주인을 비교) */
var PG_SHEET = '프리시즌_추측';
function pgSheet_(){
  var sh = SS.getSheetByName(PG_SHEET);
  if(!sh){ sh = SS.insertSheet(PG_SHEET); sh.appendRow(['일시','드래프트 날짜','이름','맞춘 수','전체']); sh.getRange('B:B').setNumberFormat('@'); sh.setFrozenRows(1); }
  return sh;
}
function psGuess_(e){
  var name = String(e.parameter.name||'').trim(), g;
  try{ g = JSON.parse(e.parameter.g||'{}'); }catch(err){ return psOut_({ok:false,msg:'json'}); }
  if(!name) return psOut_({ok:false,msg:'name'});
  var d = psLoad_();
  if(!d || !d.pool || !d.revealed) return psOut_({ok:false,msg:'not_revealed'});
  var ok = 0, tried = 0;
  d.pool.forEach(function(x){ if(g[x.id]){ tried++; if(String(g[x.id]).trim() === x.name) ok++; } });
  if(!tried) return psOut_({ok:false,msg:'empty'});
  if(!d.started){ d.started = qzTodayKST_(); psSave_(d); }   // 예전에 시작된 드래프트는 첫 제출일로 고정
  var date = d.started;
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try{
    var sh = pgSheet_(), v = sh.getDataRange().getValues();
    for(var i=1;i<v.length;i++){ if(String(v[i][2]).trim() === name && stDateStr_(v[i][1]) === date) return psOut_({ok:true, dup:true, score:parseInt(v[i][3],10)||0, total:parseInt(v[i][4],10)||0}); }
    sh.appendRow([new Date(), date, name, ok, d.pool.length]);
    return psOut_({ok:true, score:ok, total:d.pool.length});
  } finally { lock.releaseLock(); }
}
function pgBoard_(){
  var sh = SS.getSheetByName(PG_SHEET), weeks = {}, tot = {}; if(!sh) return { weeks:[], total:[] };
  sh.getDataRange().getValues().slice(1).forEach(function(r){
    var date = stDateStr_(r[1]), n = String(r[2]||'').trim(), ok = parseInt(r[3],10)||0, all = parseInt(r[4],10)||0; if(!date || !n) return;
    (weeks[date] = weeks[date] || []).push({ name:n, ok:ok, total:all });
    var T = tot[n] = tot[n] || { name:n, ok:0, total:0, n:0, best:0 }; T.ok += ok; T.total += all; T.n++; if(ok > T.best) T.best = ok;
  });
  return { weeks: Object.keys(weeks).sort().map(function(d){ return { date:d, list: weeks[d].sort(function(a,b){ return b.ok - a.ok || a.name.localeCompare(b.name,'ko'); }) }; }),
           total: Object.keys(tot).map(function(n){ return tot[n]; }).sort(function(a,b){ return b.ok - a.ok || a.n - b.n || a.name.localeCompare(b.name,'ko'); }) };
}
// 내 결과 화면용: 본인 이름의 배지만. 익명 카드가 공개되기 전(드래프트 진행 중)에는 보내지 않음
function psMyPeer_(e){
  var name = String(e.parameter.name||'').replace(/\s+/g,''), d = psLoad_();
  if(d && d.status && d.status !== 'idle' && !d.revealed) return psOut_({ok:true, hidden:true});
  return psOut_({ok:true, peer: name ? pvBadge_(pvTally_(), name) : ''});
}
function psRosterPub_(e){
  var name = String(e.parameter.name||'').trim(), today = qzTodayKST_(), voted = false, sh = SS.getSheetByName(PV_SHEET);
  if(name && sh) voted = sh.getDataRange().getValues().slice(1).some(function(r){ return String(r[2]).trim() === name && stDateStr_(r[1]) === today; });
  return psOut_({ roster: psRoster_(), voted: voted });
}
function psVote_(e){
  var p = e.parameter, name = String(p.name||'').trim(), votes;
  try{ votes = JSON.parse(p.votes||'{}'); }catch(err){ return psOut_({ok:false,msg:'json'}); }
  if(!name) return psOut_({ok:false,msg:'name'});
  var roster = psRoster_();
  if(roster.length && roster.indexOf(name) < 0) return psOut_({ok:false,msg:'not_roster'});
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try{
    var sh = pvSheet_(), today = qzTodayKST_(), v = sh.getDataRange().getValues(), n = 0;
    for(var i=v.length-1;i>=1;i--){ if(String(v[i][2]).trim() === name && stDateStr_(v[i][1]) === today) sh.deleteRow(i+1); }
    PV_CATS.forEach(function(c){
      var seen = {}, picks = (votes[c]||[]).map(function(x){ return String(x||'').trim(); })
        .filter(function(x){ if(!x || x === name || seen[x] || (roster.length && roster.indexOf(x) < 0)) return false; seen[x] = 1; return true; }).slice(0,2);
      if(picks.length){ sh.appendRow([new Date(), today, name, c, picks[0], picks[1]||'']); n += picks.length; }
    });
    return psOut_({ok:true, count:n});
  } finally { lock.releaseLock(); }
}

function psSubmit_(e){
  var p = e.parameter, name = String(p.name||'').trim(), code = String(p.code||'').trim().toUpperCase();
  if(!name || !/^([FMHA][EP][BC][GTWI]|[AD][SL][EP][BC]([TM][VQ])?)$/.test(code)) return psOut_({ok:false,msg:'invalid'});
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try{
    var d = psLoad_();
    var roster = psRoster_();
    if(roster.length && roster.indexOf(name) < 0) return psOut_({ok:false,msg:'not_roster'});
    if(d && d.status !== 'idle'){
      // 늦은 합류: 드래프트 진행 중(또는 다 뽑았지만 공개 전)이면 카드를 풀 맨 뒤에 추가하고 주장 차례를 한 번 늘림
      if(d.revealed || !d.pool) return psOut_({ok:false,msg:'draft_started'});
      if(d.captains.some(function(c){ return c.name === name; })) return psOut_({ok:false,msg:'captain'});
      if(d.pool.some(function(x){ return x.name === name; })) return psOut_({ok:false,msg:'already'});
      var gkL = p.gk==='1', scL = {}; try{ scL = JSON.parse(p.scores||'{}'); }catch(err){}
      d.pool.push({ id: Utilities.getUuid().slice(0,8), no: d.pool.length+1, name:name, code:code, scores:scL, gk:gkL, ans:String(p.ans||''),
                    peer: pvBadge_(pvTally_(), name), pick:null, pickNo:null, late:true });
      d.order = psSnake_(d.pool.length, 3);   // 앞쪽 순서는 그대로, 맨 끝에 한 차례 추가
      if(d.status === 'done') d.status = 'drafting';
      psSave_(d);
      psSheet_().appendRow([new Date(), name, code, String(p.scores||'{}'), gkL, String(p.ans||'')]);
      return psOut_({ok:true, late:true});
    }
    var sh = psSheet_(), v = sh.getDataRange().getValues(), row = -1;
    for(var i=1;i<v.length;i++){ if(String(v[i][1]).trim()===name){ row=i+1; break; } }
    if(row>0) return psOut_({ok:false,msg:'already'});   // 최초 1회만 (재검사는 운영자가 결과 삭제 후)
    sh.appendRow([new Date(), name, code, String(p.scores||'{}'), p.gk==='1', String(p.ans||'')]);
    return psOut_({ok:true});
  } finally { lock.releaseLock(); }
}

function psState_(e){
  var d = psLoad_(), admin = e.parameter.admin === PS_ADMIN_PW;
  var roster = psRoster_();
  var entries = psEntries_().filter(function(x){ return !roster.length || roster.indexOf(x.name) >= 0; });
  if(!d || d.status === 'idle'){
    var done = entries.map(function(x){ return x.name; });
    return psOut_({ status:'idle', count: entries.length, admin: admin, roster: roster, submitted: done,
      names: admin ? entries.map(function(x){ return x.name+' ('+x.code+')'; }) : undefined });
  }
  var pool = d.pool.map(function(x){
    var o = { id:x.id, no:x.no, code:x.code, scores:x.scores, gk:x.gk, ans:x.ans||'', pick:x.pick, pickNo:x.pickNo, peer:x.peer||'', late:!!x.late };
    if(d.revealed || admin) o.name = x.name;
    return o;
  });
  var turn = d.picks < d.order.length ? d.order[d.picks] : -1;
  var out = { status:d.status, revealed:!!d.revealed, started:d.started||'', captains:d.captains.map(function(c){ return c.name; }),
    pool:pool, turn:turn, pickCount:d.picks, total:d.order.length, admin:admin };
  if(admin) out.pins = d.captains.map(function(c){ return c.pin; });
  // 이름 맞추기용: 참가 선수 이름 목록(가나다순 · 카드와의 연결 정보 없음)
  out.nameList = d.pool.map(function(x){ return x.name; }).sort(function(a,b){ return a.localeCompare(b,'ko'); });
  return psOut_(out);
}

function psSetup_(e){
  if(e.parameter.admin !== PS_ADMIN_PW) return psOut_({ok:false,msg:'pw'});
  var caps = String(e.parameter.captains||'').split(',').map(function(s){ return s.trim(); }).filter(String);
  if(caps.length !== 3) return psOut_({ok:false,msg:'need3'});
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try{
    var roster = psRoster_();
    var entries = psEntries_().filter(function(x){ return caps.indexOf(x.name) < 0 && (!roster.length || roster.indexOf(x.name) >= 0); });
    for(var i=entries.length-1;i>0;i--){ var j=Math.floor(Math.random()*(i+1)); var t=entries[i]; entries[i]=entries[j]; entries[j]=t; }
    var tally = pvTally_();   // 드래프트 시작 시점까지 모인 동료 추천으로 배지 결정
    var pool = entries.map(function(x,i){ return { id: Utilities.getUuid().slice(0,8), no:i+1, name:x.name, code:x.code, scores:x.scores, gk:x.gk, ans:x.ans, peer:pvBadge_(tally, x.name), pick:null, pickNo:null }; });
    var d = { status:'drafting', revealed:false, picks:0, started: qzTodayKST_(),
      captains: caps.map(function(n){ return { name:n, pin: String(Math.floor(1000+Math.random()*9000)) }; }),
      pool: pool, order: psSnake_(pool.length, 3) };
    psSave_(d);
    return psOut_({ok:true, pins:d.captains.map(function(c){ return c.pin; }), count:pool.length});
  } finally { lock.releaseLock(); }
}

function psPick_(e){
  var p = e.parameter, ci = parseInt(p.cap,10);
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try{
    var d = psLoad_();
    if(!d || d.status !== 'drafting') return psOut_({ok:false,msg:'not_drafting'});
    var turn = d.order[d.picks];
    if(ci !== turn) return psOut_({ok:false,msg:'not_your_turn'});
    if(String(p.pin) !== d.captains[ci].pin) return psOut_({ok:false,msg:'pin'});
    var x = d.pool.filter(function(q){ return q.id === p.id; })[0];
    if(!x || x.pick !== null) return psOut_({ok:false,msg:'taken'});
    x.pick = ci; d.picks++; x.pickNo = d.picks;
    if(d.picks >= d.order.length) d.status = 'done';
    psSave_(d);
    return psOut_({ok:true, no:x.no});
  } finally { lock.releaseLock(); }
}

// 운영자: op=reveal | undo | reset(드래프트만 초기화) | wipe(검사 결과까지 전부 삭제)
function psAdmin_(e){
  if(e.parameter.admin !== PS_ADMIN_PW) return psOut_({ok:false,msg:'pw'});
  var op = e.parameter.op;
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try{
    var d = psLoad_();
    if(op === 'reveal' && d){ d.revealed = true; psSave_(d); }
    else if(op === 'hide' && d){ d.revealed = false; psSave_(d); }
    else if(op === 'undo' && d && d.picks > 0){
      var last = d.pool.filter(function(q){ return q.pickNo === d.picks; })[0];
      if(last){ last.pick = null; last.pickNo = null; }
      d.picks--; d.status = 'drafting'; psSave_(d);
    }
    else if(op === 'roster'){
      var names = String(e.parameter.names||'').split(/[,\n]/).map(function(s){ return s.trim(); }).filter(String);
      PropertiesService.getScriptProperties().setProperty('PS_ROSTER', JSON.stringify(names));
    }
    else if(op === 'delone'){
      var nm = String(e.parameter.name||'').trim(), sh1 = psSheet_(), v1 = sh1.getDataRange().getValues();
      for(var k=v1.length-1;k>=1;k--){ if(String(v1[k][1]).trim()===nm) sh1.deleteRow(k+1); }
    }
    else if(op === 'reset'){ psSave_({status:'idle'}); }
    else if(op === 'wipe'){ psSave_({status:'idle'}); var sh=psSheet_(); if(sh.getLastRow()>1) sh.deleteRows(2, sh.getLastRow()-1); }
    else return psOut_({ok:false,msg:'op'});
    return psOut_({ok:true});
  } finally { lock.releaseLock(); }
}


/* ═══════════════ 프리시즌 대시보드: 개인 승점 (→ 정규리그 출발 승점으로 이어짐) ═══════════════
 * 스타리그 기록지와 같은 모양의 시트 2개 (없으면 자동 생성 · 시트에 직접 써도 되고, 페이지에서 입력해도 같은 칸에 기록됨)
 *  프리시즌_출석     [날짜, 블루, 옐로, 레드]          ← 팀 칸에 그날 뛴 사람 이름을 띄어쓰기로 나열 (첫 번째 = 주장)
 *  프리시즌_경기결과 [날짜, 라운드, 쉬는팀, 블루, 옐로, 레드]  ← 종이 기록지와 같은 모양
 *      쉬는팀 칸에 블루/옐로/레드 중 하나 → 나머지 두 팀이 경기 (득점 없는 팀 칸은 비워도 0골)
 *      팀 칸에는 득점자(도움) 을 띄어쓰기로 나열   예) 이호진(김동우) 이호진 권호연 → 3골 (첫 골은 김동우 도움)
 *      자책골 = 득점한 팀 칸에 '자책골'(또는 OG) · @2 = 이 팀이 2실점 안고 시작
 *      (쉬는팀 칸을 비우면 스타리그 방식: 빈칸인 팀이 쉬는 팀, 무득점은 0)
 *      이름에는 띄어쓰기를 넣지 않음 (폴 허 → 폴허)
 * 개인 승점: 그 라운드에 내 팀이 이기면 +3 · 비기면 +1 · 지면 0 (쉬는 라운드는 0)
 */
var PB_ATT = '프리시즌_출석', PB_GAME = '프리시즌_경기결과';
var PB_OLD_TEAM = '프리시즌_팀', PB_OLD_ROUND = '프리시즌_라운드', PB_OLD_GOAL = '프리시즌_골';   // 예전 형식 (자동 변환 후 삭제)
var PB_KEYS = ['blue','yellow','red'], PB_KO = { blue:'블루', yellow:'옐로', red:'레드' };
var PB_COL = { blue:'#1f6fe5', yellow:'#e39a00', red:'#e0392f' };
var PB_WIN = 3, PB_DRAW = 1, PB_LOSS = 0;
// 몸값(단위: 만 BP · 현재 FC온라인 시세처럼 최상위가 1억 BP 안팎이 되도록) — 0에서 시작
//  clean = 무실점 라운드(팀원 전원, 골과 같은 값) · mvp = 그날 가장 많이 번 사람 추가 보너스(동률이면 모두)
var PB_VAL = { att:100, captain:10, win:60, draw:20, goal:60, assist:30, clean:60, mvp:200, absent:50, absentRate:0.1, dormantAt:4, dormantRate:0.5, cap:99999 };
//  결석: 그때 BP의 10%(최소 50만) 차감 · 4번 연속 결석부터는 매번 절반(50%) 차감   // captain = 그날 팀을 꾸린 임시 주장(출석 칸 맨 앞 이름)

function pbSheet_(name, header, widths){
  var sh = SS.getSheetByName(name);
  if(!sh){
    sh = SS.insertSheet(name);
    sh.appendRow(header);
    sh.getRange(1,1,1,header.length).setFontWeight('bold').setBackground('#0b1f4b').setFontColor('#ffffff').setHorizontalAlignment('center');
    PB_KEYS.forEach(function(k, i){ sh.getRange(1, header.length-2+i).setBackground(PB_COL[k]); });
    sh.setFrozenRows(1);
    sh.getRange('A:A').setNumberFormat('@');           // 날짜는 글자 그대로 (자동 변환 방지)
    sh.getRange(1,1,sh.getMaxRows(),header.length).setWrap(true).setVerticalAlignment('middle');
    (widths||[]).forEach(function(w,i){ sh.setColumnWidth(i+1, w); });
  }
  return sh;
}
function pbSheets_(){
  var S = {
    att:  pbSheet_(PB_ATT,  ['날짜','블루','옐로','레드'], [100,280,280,280]),
    game: pbSheet_(PB_GAME, ['날짜','라운드','쉬는팀','블루','옐로','레드'], [100,60,70,260,260,260])
  };
  // 예전 5칸 형식(쉬는팀 칸 없음)이면 칸을 끼워 넣음
  if(String(S.game.getRange(1,3).getValue()).trim() !== '쉬는팀'){
    S.game.insertColumnAfter(2); S.game.getRange(1,3).setValue('쉬는팀').setBackground('#0b1f4b'); S.game.setColumnWidth(3,70);
  }
  pbMigrate_(S);
  return S;
}
function pbSetupSheets(){ pbSheets_(); Logger.log('프리시즌 시트 2개 준비 완료'); }

function pbKey_(v){
  var t = String(v||'').trim().toLowerCase();
  if(t==='블루'||t==='blue'||t==='파랑'||t==='b') return 'blue';
  if(t==='옐로'||t==='옐로우'||t==='yellow'||t==='노랑'||t==='y') return 'yellow';
  if(t==='레드'||t==='red'||t==='빨강'||t==='r') return 'red';
  return '';
}
// 날짜를 어떻게 적어도 yyyy-MM-dd 로 맞춤: 20261114 · 2026.11.14 · 2026/11/14 · 2026-11-14 · 26.11.14 · 날짜 서식 칸
function pbDate_(v){
  if(v instanceof Date) return Utilities.formatDate(v, QZ_TZ, 'yyyy-MM-dd');
  var t = String(v === undefined || v === null ? '' : v).trim(), m;
  if((m = t.match(/^(\d{4})(\d{2})(\d{2})$/))) return m[1]+'-'+m[2]+'-'+m[3];
  if((m = t.match(/^(\d{2}|\d{4})\s*[.\/-]\s*(\d{1,2})\s*[.\/-]\s*(\d{1,2})\.?$/)))
    return (m[1].length === 2 ? '20'+m[1] : m[1]) + '-' + ('0'+m[2]).slice(-2) + '-' + ('0'+m[3]).slice(-2);
  return t;
}
function pbName_(v){ return String(v||'').replace(/[\s()@]+/g,''); }   // 이름 안의 띄어쓰기·괄호 제거 (칸 안에서 띄어쓰기가 구분자)
function pbRows_(sh){ var v = sh.getDataRange().getValues(); return v.length > 1 ? v.slice(1) : []; }
function pbDeleteWhere_(sh, pred){
  var v = sh.getDataRange().getValues();
  for(var i=v.length-1;i>=1;i--){ if(pred(v[i])) sh.deleteRow(i+1); }
}
// 골 목록 → 칸에 쓸 글자 ("득점자(도움) 득점자" / 무득점은 0)
function pbCell_(goals, handicap){
  var t = goals.map(function(g){
    var s = pbName_(g.scorer), a = pbName_(g.assist);
    if(!s || s === '자책골' || s === 'OG') return '자책골';
    return a ? s + '(' + a + ')' : s;
  });
  if(handicap) t.push('@' + handicap);
  return t.length ? t.join(' ') : '0';
}
// 칸 글자 → { goals:[{scorer,assist}], handicap }
function pbParse_(s){
  var goals = [], handicap = 0;
  String(s === undefined || s === null ? '' : s).trim().split(/\s+/).forEach(function(tok){
    if(!tok || tok === '0' || tok === '대기') return;
    var hm = tok.match(/^@(\d+)$/); if(hm){ handicap += parseInt(hm[1],10); return; }
    if(tok === '자책골' || tok === 'OG'){ goals.push({ scorer:'자책골', assist:'' }); return; }
    var m = tok.match(/^([^(]+)(?:\(([^)]*)\))?$/); if(!m) return;
    goals.push({ scorer:m[1].trim(), assist:(m[2]||'').trim() });
  });
  return { goals:goals, handicap:handicap };
}
// 예전 3개 시트(프리시즌_팀/라운드/골)가 남아 있으면 새 형식으로 옮기고 지움 — 한 번만 실행됨
function pbMigrate_(S){
  var ot = SS.getSheetByName(PB_OLD_TEAM), or_ = SS.getSheetByName(PB_OLD_ROUND), og = SS.getSheetByName(PB_OLD_GOAL);
  if(!ot && !or_ && !og) return;
  if(ot){
    var byDate = {}, order = [];
    pbRows_(ot).forEach(function(r){
      var d = pbDate_(r[0]), k = pbKey_(r[1]), n = pbName_(r[2]); if(!d || !k || !n) return;
      if(!byDate[d]){ byDate[d] = { blue:[], yellow:[], red:[] }; order.push(d); }
      if(String(r[3]||'').trim() === '주장') byDate[d][k].unshift(n); else byDate[d][k].push(n);
    });
    var have = {}; pbRows_(S.att).forEach(function(r){ have[pbDate_(r[0])] = 1; });
    order.forEach(function(d){ if(!have[d]) S.att.appendRow([d].concat(PB_KEYS.map(function(k){ return byDate[d][k].join(' '); }))); });
  }
  if(or_){
    var gl = {};
    if(og) pbRows_(og).forEach(function(r){
      var key = pbDate_(r[0]) + '|' + parseInt(r[1],10) + '|' + pbKey_(r[2]);
      (gl[key] = gl[key] || []).push({ scorer:r[3], assist:r[4] });
    });
    var haveR = {}; pbRows_(S.game).forEach(function(r){ haveR[pbDate_(r[0]) + '|' + parseInt(r[1],10)] = 1; });
    pbRows_(or_).forEach(function(r){
      var d = pbDate_(r[0]), rn = parseInt(r[1],10), a = pbKey_(r[2]), b = pbKey_(r[3]);
      if(!d || !rn || !a || !b || haveR[d+'|'+rn]) return;
      var sc = {}; sc[a] = parseInt(r[4],10)||0; sc[b] = parseInt(r[5],10)||0;
      var rest0 = PB_KEYS.filter(function(k){ return k !== a && k !== b; })[0];
      S.game.appendRow([d, rn, PB_KO[rest0]].concat(PB_KEYS.map(function(k){
        if(k !== a && k !== b) return '';
        var g = (gl[d+'|'+rn+'|'+k] || []).slice();
        while(g.length < sc[k]) g.push({ scorer:'자책골', assist:'' });   // 득점자 없이 점수만 있던 골
        return pbCell_(g);
      })));
    });
  }
  [ot, or_, og].forEach(function(sh){ if(sh) SS.deleteSheet(sh); });
}

function pbCompute_(){
  var S = pbSheets_(), days = {}, players = {};
  function day(d){ if(!days[d]) days[d] = { date:d, teams:{blue:[],yellow:[],red:[]}, rounds:[] }; return days[d]; }
  function pl(n){ if(!players[n]) players[n] = { name:n, pts:0, r:0, w:0, d:0, l:0, g:0, a:0, days:0, last:'', val:0, dv:0, cs:0, mvp:0, miss:0 }; return players[n]; }

  pbRows_(S.att).forEach(function(r){
    var d = pbDate_(r[0]); if(!d) return;
    PB_KEYS.forEach(function(k, ci){
      String(r[ci+1]||'').trim().split(/\s+/).forEach(function(n){
        if(!n) return;
        var T = day(d).teams[k];
        if(!T.some(function(m){ return m.name === n; })) T.push({ name:n, role: T.length ? '선수' : '주장' });
      });
    });
  });
  pbRows_(S.game).forEach(function(r){
    var d = pbDate_(r[0]), rn = parseInt(r[1],10); if(!d || !rn) return;
    var playing = [], parsed = {}, rest = pbKey_(r[2]);
    PB_KEYS.forEach(function(k, ci){
      var raw = r[ci+3], cell = String(raw === undefined || raw === null ? '' : raw).trim();
      if(rest ? k === rest : (cell === '' || cell === '대기')) return;
      playing.push(k); parsed[k] = pbParse_(cell);
    });
    if(playing.length !== 2) return;                    // 한 라운드는 두 팀 경기 (나머지 한 팀은 빈칸)
    var a = playing[0], b = playing[1], goals = [];
    parsed[a].goals.forEach(function(g){ goals.push({ team:a, scorer:g.scorer, assist:g.assist }); });
    parsed[b].goals.forEach(function(g){ goals.push({ team:b, scorer:g.scorer, assist:g.assist }); });
    day(d).rounds.push({ r:rn, a:a, b:b,
      sa: parsed[a].goals.length + parsed[b].handicap, sb: parsed[b].goals.length + parsed[a].handicap,
      ha: parsed[a].handicap, hb: parsed[b].handicap, goals:goals });
  });

  var list = Object.keys(days).sort().map(function(d){
    var D = days[d]; D.rounds.sort(function(x,y){ return x.r - y.r; });
    var tb = {}; PB_KEYS.forEach(function(k){ tb[k] = { team:k, p:0, w:0, d:0, l:0, gf:0, ga:0, pts:0 }; });
    var gained = {};   // 그날 개인 승점
    var dv = {};       // 그날 몸값 변동(억)
    Object.keys(players).forEach(function(n){ var P0 = players[n]; P0.miss++;
      dv[n] = P0.miss >= PB_VAL.dormantAt ? -Math.round(P0.val * PB_VAL.dormantRate) : -Math.max(PB_VAL.absent, Math.round(P0.val * PB_VAL.absentRate)); });   // 전에 나온 적 있는 사람은 일단 결석 처리 → 출석이면 아래에서 덮어씀
    PB_KEYS.forEach(function(k){ D.teams[k].forEach(function(m){ var P = pl(m.name); P.days++; P.miss = 0; if(d > P.last) P.last = d; gained[m.name] = 0; dv[m.name] = PB_VAL.att + (m.role === '주장' ? PB_VAL.captain : 0); }); });
    D.rounds.forEach(function(R){
      var ra = R.sa > R.sb ? 'w' : R.sa < R.sb ? 'l' : 'd', rb = ra==='w' ? 'l' : ra==='l' ? 'w' : 'd';
      [[R.a, ra, R.sa, R.sb],[R.b, rb, R.sb, R.sa]].forEach(function(x){
        var t = tb[x[0]], pt = x[1]==='w' ? PB_WIN : x[1]==='d' ? PB_DRAW : PB_LOSS;
        t.p++; t[x[1]]++; t.gf += x[2]; t.ga += x[3]; t.pts += pt;
        var vv = (x[1]==='w' ? PB_VAL.win : x[1]==='d' ? PB_VAL.draw : 0) + (x[3] === 0 ? PB_VAL.clean : 0);   // x[3] = 실점
        D.teams[x[0]].forEach(function(m){ var P = pl(m.name); P.r++; P[x[1]]++; P.pts += pt; gained[m.name] += pt; dv[m.name] += vv; if(x[3] === 0) P.cs++; });
      });
      R.goals.forEach(function(g){
        if(g.scorer && g.scorer !== '자책골' && players[g.scorer]){ players[g.scorer].g++; if(gained[g.scorer] !== undefined) dv[g.scorer] += PB_VAL.goal; }
        if(g.assist && players[g.assist]){ players[g.assist].a++; if(gained[g.assist] !== undefined) dv[g.assist] += PB_VAL.assist; }
      });
    });
    D.table = PB_KEYS.map(function(k){ return tb[k]; }).sort(function(x,y){ return y.pts - x.pts || (y.gf-y.ga) - (x.gf-x.ga) || y.gf - x.gf; });
    D.gained = gained;
    // 오늘의 MVP: 그날 출석자 중 가장 많이 번 사람 (라운드 기록이 있는 날만)
    D.mvp = [];
    if(D.rounds.length){
      var best = 0; Object.keys(gained).forEach(function(n){ if(dv[n] > best) best = dv[n]; });
      if(best > PB_VAL.att) Object.keys(gained).forEach(function(n){ if(dv[n] === best) D.mvp.push(n); });
      D.mvp.forEach(function(n){ dv[n] += PB_VAL.mvp; players[n].mvp++; });
    }
    D.val = {};
    Object.keys(dv).forEach(function(n){
      var P = players[n], nv = Math.min(PB_VAL.cap, Math.max(0, P.val + dv[n]));
      D.val[n] = nv - P.val; P.dv = nv - P.val; P.val = nv;
    });
    return D;
  });
  var ps = Object.keys(players).map(function(n){ return players[n]; })
    .sort(function(x,y){ return y.val - x.val || y.pts - x.pts || y.w - x.w || (y.g+y.a) - (x.g+x.a) || x.name.localeCompare(y.name,'ko'); });
  return { days:list, players:ps, rule:{ win:PB_WIN, draw:PB_DRAW, loss:PB_LOSS, val:PB_VAL } };
}

function pbBoard_(e){
  var out = pbCompute_();
  out.admin = e.parameter.admin === PS_ADMIN_PW;
  out.today = qzTodayKST_();
  out.guess = pgBoard_();
  // 팀 편성 편의: 방금 끝난 드래프트 결과 (운영자에게만 실명)
  if(out.admin){
    var d = psLoad_();
    if(d && d.pool && d.status === 'done'){
      out.draft = d.captains.map(function(c, i){
        return [c.name].concat(d.pool.filter(function(x){ return x.pick === i; }).sort(function(a,b){ return a.pickNo-b.pickNo; }).map(function(x){ return x.name; })).map(pbName_);
      });
    }
    out.roster = psRoster_();
  }
  return psOut_(out);
}

// 운영자: 그날 출석(팀 편성) 저장 — teams = {"blue":["주장","선수"...],"yellow":[...],"red":[...]} (첫 번째가 주장)
function pbTeams_(e){
  if(e.parameter.admin !== PS_ADMIN_PW) return psOut_({ok:false,msg:'pw'});
  var date = String(e.parameter.date||'').trim(), teams;
  try{ teams = JSON.parse(e.parameter.teams||'{}'); }catch(err){ return psOut_({ok:false,msg:'json'}); }
  if(!/^\d{4}-\d{2}-\d{2}$/.test(date)) return psOut_({ok:false,msg:'date'});
  var lock = LockService.getScriptLock(); lock.waitLock(15000);
  try{
    var sh = pbSheets_().att, count = 0;
    var cells = PB_KEYS.map(function(k){
      var seen = {}, list = [];
      (teams[k]||[]).forEach(function(n){ n = pbName_(n); if(n && !seen[n]){ seen[n] = 1; list.push(n); } });
      count += list.length; return list.join(' ');
    });
    pbDeleteWhere_(sh, function(r){ return pbDate_(r[0]) === date; });
    if(count) sh.appendRow([date].concat(cells));
    return psOut_({ok:true, count:count});
  } finally { lock.releaseLock(); }
}

// 운영자: 한 라운드 저장(덮어쓰기)·삭제 — goals = [{"team":"blue","scorer":"이름","assist":"이름"}...]
//  goals 를 보내면 득점은 골 개수로 계산, 안 보내면 sa/sb 값만큼 '자책골'(득점자 없음)로 채움
function pbRound_(e){
  var p = e.parameter;
  if(p.admin !== PS_ADMIN_PW) return psOut_({ok:false,msg:'pw'});
  var date = String(p.date||'').trim(), rn = parseInt(p.round,10), a = pbKey_(p.a), b = pbKey_(p.b);
  if(!/^\d{4}-\d{2}-\d{2}$/.test(date) || !rn) return psOut_({ok:false,msg:'bad'});
  var lock = LockService.getScriptLock(); lock.waitLock(15000);
  try{
    var S = pbSheets_();
    pbDeleteWhere_(S.game, function(r){ return pbDate_(r[0]) === date && parseInt(r[1],10) === rn; });
    if(p.op === 'del') return psOut_({ok:true});
    if(!a || !b || a === b) return psOut_({ok:false,msg:'teams'});
    var goals = []; try{ goals = JSON.parse(p.goals||'[]'); }catch(err){ goals = []; }
    var by = {}; by[a] = []; by[b] = [];
    if(goals.length){ goals.forEach(function(g){ var k = pbKey_(g.team); if(by[k]) by[k].push(g); }); }
    else {
      for(var i=0;i<(parseInt(p.sa,10)||0);i++) by[a].push({ scorer:'자책골' });
      for(var j=0;j<(parseInt(p.sb,10)||0);j++) by[b].push({ scorer:'자책골' });
    }
    var restK = PB_KEYS.filter(function(k){ return !by[k]; })[0];
    S.game.appendRow([date, rn, PB_KO[restK]].concat(PB_KEYS.map(function(k){ return by[k] ? pbCell_(by[k]) : ''; })));
    // 날짜·라운드 순으로 정렬 (기록지처럼 위에서 아래로 읽히게)
    if(S.game.getLastRow() > 2) S.game.getRange(2,1,S.game.getLastRow()-1,6).sort([{column:1,ascending:true},{column:2,ascending:true}]);
    return psOut_({ok:true, sa:by[a].length, sb:by[b].length});
  } finally { lock.releaseLock(); }
}
